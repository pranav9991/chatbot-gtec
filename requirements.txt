from pipecat.processors.base import FrameProcessor
from pipecat.frames import TextFrame

INTERRUPTS = {"stop", "cancel", "wait"}

class InterruptProcessor(FrameProcessor):
    async def process(self, frame, direction):
        if isinstance(frame, TextFrame):
            if any(word in frame.text.lower() for word in INTERRUPTS):
                await self.pipeline.flush()
                return None
        return frame
