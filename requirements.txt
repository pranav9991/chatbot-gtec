import streamlit as st
import cv2
import numpy as np
from PIL import Image
from mtcnn import MTCNN

# Initialize MTCNN face detector
detector = MTCNN()

# Heuristic behaviour detection
def detect_behaviour(eye_img, lip_img):
    # Eye darkness heuristic
    eye_gray = cv2.cvtColor(eye_img, cv2.COLOR_BGR2GRAY)
    eye_darkness = np.mean(eye_gray)

    # Lip darkness using LAB color space
    lip_lab = cv2.cvtColor(lip_img, cv2.COLOR_BGR2LAB)
    L_channel = lip_lab[:, :, 0]  # lightness
    lip_darkness = np.mean(L_channel)

    if lip_darkness < 60:   # darker lips threshold
        return "Possible Lip Biting / Dark Lips"
    elif eye_darkness < 60:  # darker eyes
        return "Possible Eye Rubbing"
    else:
        return "No clear behaviour detected"

# Helper: crop a region around a keypoint
def crop_region(img, center, size=50):
    x, y = center
    x1 = max(0, int(x - size//2))
    y1 = max(0, int(y - size//2))
    x2 = min(img.shape[1], int(x + size//2))
    y2 = min(img.shape[0], int(y + size//2))
    return img[y1:y2, x1:x2]

# Streamlit UI
st.title("Facial Behaviour Detection (Lip / Eye Analysis)")
uploaded = st.file_uploader("Upload a face image", type=["jpg", "jpeg", "png"])

if uploaded:
    image = Image.open(uploaded).convert("RGB")
    img_np = np.array(image)
    img_bgr = cv2.cvtColor(img_np, cv2.COLOR_RGB2BGR)

    results = detector.detect_faces(cv2.cvtColor(img_bgr, cv2.COLOR_BGR2RGB))
    if not results:
        st.error("No face detected.")
    else:
        # Take the first detected face
        keypoints = results[0]['keypoints']
        left_eye = keypoints['left_eye']
        right_eye = keypoints['right_eye']
        mouth_left = keypoints['mouth_left']
        mouth_right = keypoints['mouth_right']

        # Crop eyes and lips
        eye_center = ((left_eye[0] + right_eye[0]) // 2, (left_eye[1] + right_eye[1]) // 2)
        eyes_crop = crop_region(img_bgr, eye_center, size=60)

        lip_center = ((mouth_left[0] + mouth_right[0]) // 2, (mouth_left[1] + mouth_right[1]) // 2)
        lips_crop = crop_region(img_bgr, lip_center, size=50)

        behaviour = detect_behaviour(eyes_crop, lips_crop)

        st.image(image, caption="Uploaded Image", use_container_width=True)
        st.subheader(f"Detected Behaviour: {behaviour}")
