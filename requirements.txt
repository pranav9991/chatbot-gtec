import streamlit as st
import cv2
import dlib
import numpy as np
from PIL import Image
import io
from deepface import DeepFace

LANDMARK_MODEL = "shape_predictor_68_face_landmarks.dat"

@st.cache_resource
def load_dlib_models():
    detector = dlib.get_frontal_face_detector()
    predictor = dlib.shape_predictor(LANDMARK_MODEL)
    return detector, predictor

def read_image(uploaded_file):
    image = Image.open(uploaded_file).convert('RGB')
    return np.array(image)[:, :, ::-1]

def detect_and_predict(img_bgr, detector, predictor):
    gray = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
    faces = detector(gray)
    output = img_bgr.copy()

    for face in faces:
        x1, y1, x2, y2 = face.left(), face.top(), face.right(), face.bottom()
        face_crop = img_bgr[y1:y2, x1:x2]

        try:
            analysis = DeepFace.analyze(face_crop, actions=['age', 'gender'], enforce_detection=False)
            age = int(analysis[0]['age'])
            gender = analysis[0]['dominant_gender'].capitalize()
        except Exception:
            age, gender = "?", "?"

        # Draw bounding box and label
        cv2.rectangle(output, (x1, y1), (x2, y2), (0,255,0), 2)
        cv2.putText(output, f"{gender}, {age}", (x1, y1-10),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0,255,0), 2)

        # Landmarks
        shape = predictor(gray, face)
        for i in range(68):
            x, y = shape.part(i).x, shape.part(i).y
            cv2.circle(output, (x, y), 1, (0, 0, 255), -1)

    return output

def main():
    st.set_page_config(page_title="Age & Gender Prediction", layout="centered")
    st.title("Age and Gender Prediction (DeepFace + dlib landmarks)")

    detector, predictor = load_dlib_models()

    uploaded = st.file_uploader("Upload an image", type=["jpg","jpeg","png"])
    cam = st.camera_input("Or take a photo")

    img = None
    if uploaded: img = read_image(uploaded)
    elif cam: img = read_image(io.BytesIO(cam.getvalue()))

    if img is not None:
        with st.spinner("Analyzing..."):
            result = detect_and_predict(img, detector, predictor)
        st.image(cv2.cvtColor(result, cv2.COLOR_BGR2RGB), use_column_width=True)

if __name__ == "__main__":
    main()
