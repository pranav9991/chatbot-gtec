from pipecat.processors.base import FrameProcessor
from pipecat.frames import TextFrame

class KTPromptProcessor(FrameProcessor):
    def __init__(self):
        super().__init__()
        self.used = False

    async def process(self, frame, direction):
        if isinstance(frame, TextFrame) and not self.used:
            self.used = True
            frame.text = f"""
You are a senior software engineer giving KT.

Rules:
- Use ONLY the provided code context
- Mention filenames and functions
- Be concise and technical
- If unsure, say so

{frame.text}
"""
        return frame

    async def on_pipeline_flush(self):
        self.used = False
